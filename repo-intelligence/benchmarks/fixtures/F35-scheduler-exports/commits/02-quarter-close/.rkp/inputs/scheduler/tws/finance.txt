# IBM Workload Scheduler: composer create finance.txt from jobs=FINWS#@ and sched=FINWS#@
$JOBS
FINWS#GL_EXTRACT
 SCRIPTNAME "/opt/fin/bin/gl_extract.sh"
 STREAMLOGON finbatch
 DESCRIPTION "Extract the general ledger"
 TASKTYPE UNIX
 RECOVERY STOP

FINWS#GL_POST
 DOCOMMAND "/opt/fin/bin/gl_post.sh -period current"
 STREAMLOGON finbatch
 DESCRIPTION "Post the day's journals"
 TASKTYPE UNIX
 RECOVERY RERUN AFTER FINWS#GL_RESET

$CALENDAR
MONTHEND
 "Last working day of each month"
 01/30/2026 02/27/2026 03/31/2026

SCHEDULE FINWS#FIN_DAILY
DESCRIPTION "Daily finance close"
ON RUNCYCLE DAILY1 "FREQ=DAILY;INTERVAL=1"
EXCEPT HOLIDAYS
AT 0400
FOLLOWS CTMWS#ORDERS_FEED.@
:
FINWS#GL_EXTRACT
FINWS#GL_POST
 FOLLOWS GL_EXTRACT
 NEEDS 1 FINWS#GL_LOCK
END

SCHEDULE FINWS#FIN_MONTHEND
DESCRIPTION "Month-end close"
ON MONTHEND
AT 2100
:
FINWS#GL_CLOSE SCRIPTNAME "/opt/fin/bin/gl_close.sh" STREAMLOGON finbatch
 FOLLOWS FINWS#FIN_DAILY.GL_POST
END

SCHEDULE FINWS#FIN_QUARTER
DESCRIPTION "Quarter close"
ON QUARTEREND
AT 2300
:
FINWS#GL_QUARTER SCRIPTNAME "/opt/fin/bin/gl_quarter.sh" STREAMLOGON finbatch
 FOLLOWS FINWS#FIN_MONTHEND.GL_CLOSE
END
